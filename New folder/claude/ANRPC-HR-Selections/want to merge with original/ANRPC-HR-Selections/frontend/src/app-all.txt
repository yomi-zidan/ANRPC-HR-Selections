"use client";

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import Image from "next/image";
import { Bell, LogOut, Menu, X } from "lucide-react";
import { clearAuthSession } from "@/lib/auth";

export default function EmployeeLayout({ children }) {
  const router = useRouter();
  const [employee, setEmployee] = useState(null);
  const [menuOpen, setMenuOpen] = useState(false);

  useEffect(() => {
    const saved = sessionStorage.getItem("employee");

    if (!saved || !sessionStorage.getItem("auth_token")) {
      router.replace("/login");
      return;
    }

    try {
      setEmployee(JSON.parse(saved));
    } catch {
      clearAuthSession("employee");
      router.replace("/login");
    }
  }, [router]);

  function logout() {
    clearAuthSession("employee");
    router.replace("/login");
  }

  if (!employee) return null;

  return (
    <div dir="rtl" className="employee-layout min-h-screen bg-[#F4F7FB]">

      {/* Header */}
      <header className="sticky top-0 z-40 border-b border-slate-200 bg-white/95 shadow-sm backdrop-blur-xl">
        <div className="flex min-h-[80px] items-center justify-between gap-4 px-5 py-3 lg:px-10">

          {/* Company */}
          <div className="flex items-center gap-3">
            <div className="flex h-12 w-12 items-center justify-center rounded-2xl border border-slate-200 bg-white p-2 shadow-sm">
              <Image src="/images/anrpc-logo.png" alt="ANRPC" width={40} height={40} className="h-full w-full object-contain" />
            </div>

            <div>
              <h1 className="text-sm font-bold text-[#163B63] sm:text-lg">
                نظام الاختيارات الوظيفية
              </h1>
              <p className="mt-1 hidden text-[11px] text-slate-500 sm:block sm:text-xs">
                شركة الإسكندرية الوطنية للتكرير والبتروكيماويات
              </p>
            </div>
          </div>

          {/* Desktop employee info */}
          <div className="hidden items-center gap-5 sm:flex">

              <p className="text-sm font-bold text-[#163B63]">
                {employee.ename}
              </p>
              <p className="mt-1 text-xs text-slate-500">
                رقم وظيفي: {employee.empn}

"use client";

import Link from "next/link";
import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { clearAuthSession } from "@/lib/auth";
import {
  ArrowRight,
  Upload,
  Trash2,
  Save,
  FileText,
  LoaderCircle,
} from "lucide-react";

const API_URL = "http://127.0.0.1:8000/api";

export default function NewSubmission() {
  const router = useRouter();

  const [employee, setEmployee] = useState(null);
  const [criteriaOptions, setCriteriaOptions] = useState([]);
  const [fileTypes, setFileTypes] = useState([]);
  const [contest, setContest] = useState(null);

  const [loadingOptions, setLoadingOptions] = useState(true);
  const [submitting, setSubmitting] = useState(false);
  const [message, setMessage] = useState("");
  const [messageType, setMessageType] = useState("error");

  const [submissionType, setSubmissionType] = useState("1");
  const [selectedCriteria, setSelectedCriteria] = useState([]);
  const [teamMembers, setTeamMembers] = useState("");
  const [files, setFiles] = useState([]);

  const [form, setForm] = useState({
    location: "",
    title: "",
    challenge: "",
    efforts: "",
    results: "",
    impact: "",
    inspiration: "",
  });

  const inputClass =
    "h-11 w-full rounded-md border border-slate-300 bg-white px-3 text-sm text-slate-700 outline-none transition focus:border-[#163B63] focus:ring-1 focus:ring-[#163B63]";

  const textareaClass =
    "min-h-24 w-full resize-y rounded-md border border-slate-300 bg-white p-3 text-sm text-slate-700 outline-none transition focus:border-[#163B63] focus:ring-1 focus:ring-[#163B63]";

  const labelClass = "mb-2 block text-sm font-semibold text-slate-600";

  // Load employee and form options
  useEffect(() => {
    const savedEmployee = sessionStorage.getItem("employee");
    const token = sessionStorage.getItem("auth_token");

    if (!savedEmployee || !token) {
      router.replace("/login");
      return;
    }

    try {
      setEmployee(JSON.parse(savedEmployee));
    } catch {
      clearAuthSession("employee");
      router.replace("/login");
      return;
    }

    loadOptions(token);
  }, [router]);

  async function loadOptions(token) {
    try {
      setLoadingOptions(true);

      const response = await fetch(
        `${API_URL}/employee/submission-options`,
        {
          method: "GET",
          headers: {
            Authorization: `Bearer ${token}`,
            Accept: "application/json",
          },
        }
      );

      const data = await response.json();

      if (response.status === 401) {
        clearAuthSession("employee");
        router.replace("/login");
        return;
      }

      if (!response.ok) {
        throw new Error(data.message || "تعذر تحميل بيانات النموذج.");
      }

      setCriteriaOptions(data.criteria || []);
      setFileTypes(data.file_types || []);
      setContest(data.contest || null);

    } catch (error) {
      setMessage(error.message || "حدث خطأ أثناء تحميل البيانات.");
      setMessageType("error");
    } finally {
      setLoadingOptions(false);
    }
  }

  function updateField(e) {
    const { name, value } = e.target;

    setForm((previous) => ({
      ...previous,
      [name]: value,
    }));
  }

  function toggleCriteria(id) {
    setSelectedCriteria((previous) =>
      previous.includes(id)
        ? previous.filter((item) => item !== id)
        : [...previous, id]
    );
  }

  function addFiles(e) {
    const selected = Array.from(e.target.files || []);
    const allowedExtensions = /\.(pdf|doc|docx|jpg|jpeg|png)$/i;

    const validFiles = selected.filter((file) =>
      allowedExtensions.test(file.name)
    );

    const invalidCount = selected.length - validFiles.length;

    const newFiles = validFiles.map((file) => ({
      id: crypto.randomUUID(),
      file,
      fileTypeId: fileTypes[0]?.Id
        ? String(fileTypes[0].Id)
        : "",
    }));

    setFiles((previous) => [...previous, ...newFiles]);

    if (invalidCount > 0) {
      setMessage(
        "تم تجاهل بعض الملفات. المسموح PDF وWord والصور فقط."
      );
      setMessageType("error");
    } else {
      setMessage("");
    }

    e.target.value = "";
  }

  function updateFileType(id, value) {
    setFiles((previous) =>
      previous.map((item) =>
        item.id === id
          ? { ...item, fileTypeId: value }
          : item
      )
    );
  }

  function removeFile(id) {
    setFiles((previous) =>
      previous.filter((item) => item.id !== id)
    );
  }

  async function handleSubmit(e) {
    e.preventDefault();

    setMessage("");

    const token = sessionStorage.getItem("auth_token");

    if (!token) {
      setMessage("انتهت جلسة الدخول. برجاء تسجيل الدخول مرة أخرى.");
      setMessageType("error");
      return;
    }

    if (selectedCriteria.length === 0) {
      setMessage("برجاء اختيار معيار ترشيح واحد على الأقل.");
      setMessageType("error");
      return;
    }

    if (submissionType === "2" && !teamMembers.trim()) {
      setMessage("برجاء إدخال أسماء أعضاء الفريق ومسمياتهم.");
      setMessageType("error");
      return;
    }

    if (files.length === 0) {
      setMessage("برجاء إرفاق ملف واحد على الأقل.");
      setMessageType("error");
      return;
    }

    if (files.some((item) => !item.fileTypeId)) {
      setMessage("برجاء اختيار نوع كل مرفق.");
      setMessageType("error");
      return;
    }

    const formData = new FormData();

    formData.append("title", form.title);
    formData.append("submission_type", submissionType);
    formData.append("location", form.location);
    formData.append("team_members", teamMembers);

    formData.append("challenge", form.challenge);
    formData.append("efforts", form.efforts);
    formData.append("results", form.results);
    formData.append("impact", form.impact);
    formData.append("inspiration", form.inspiration);

    selectedCriteria.forEach((id) => {
      formData.append("criteria[]", String(id));
    });

    files.forEach((item, index) => {
      formData.append(
        `files[${index}][file_type_id]`,
        String(item.fileTypeId)
      );

      formData.append(
        `files[${index}][file]`,
        item.file
      );
    });

    try {
      setSubmitting(true);

      const response = await fetch(
        `${API_URL}/employee/submissions`,
        {
          method: "POST",
          headers: {
            Authorization: `Bearer ${token}`,
            Accept: "application/json",
          },
          body: formData,
        }
      );

      const data = await response.json();

      if (response.status === 401) {
        clearAuthSession("employee");
        router.replace("/login");
        return;
      }

      if (!response.ok) {
        const validationErrors = data.errors
          ? Object.values(data.errors).flat().join(" ")
          : data.message;

        throw new Error(
          validationErrors || "تعذر حفظ المشاركة."
        );
      }

      setMessage(data.message || "تم حفظ المشاركة بنجاح.");
      setMessageType("success");

      setTimeout(() => {
        router.push("/employee/submissions");
      }, 1200);

    } catch (error) {
      setMessage(error.message || "حدث خطأ أثناء حفظ المشاركة.");
      setMessageType("error");
    } finally {
      setSubmitting(false);
    }
  }

  if (loadingOptions && !employee) {
    return null;
  }

  return (
    <div dir="rtl" className="mx-auto max-w-6xl space-y-6">

      {/* Page heading */}
      <div className="flex items-center gap-3">
        <Link
          href="/employee/submissions"
          className="rounded-lg border border-slate-200 bg-white p-2 text-[#163B63] hover:bg-slate-50"
        >
          <ArrowRight size={20} />
        </Link>

        <div>
          <h1 className="text-2xl font-bold text-[#163B63]">
            إضافة مشاركة جديدة
          </h1>
          <p className="mt-1 text-sm text-slate-500">
            نموذج المشاركة في قصص النجاح والتميز
          </p>
        </div>
      </div>

      {contest && (
        <div className="rounded-lg border border-blue-100 bg-blue-50 p-4 text-sm text-[#163B63]">
          <p className="font-bold">{contest.ContestName}</p>
          <p className="mt-1">
            آخر موعد للتقديم:{" "}
            {new Date(contest.DeadlineDate).toLocaleDateString("ar-EG")}
          </p>
        </div>
      )}

      {loadingOptions && (
        <div className="flex items-center gap-2 rounded-lg bg-white p-4 text-sm text-slate-600">
          <LoaderCircle className="animate-spin" size={18} />
          جاري تحميل معايير الترشيح وأنواع المرفقات...
        </div>
      )}

      <form onSubmit={handleSubmit} className="space-y-5">

        {/* Participant information */}
        <section className="rounded-xl border border-slate-200 bg-white p-5 shadow-sm md:p-6">
          <h2 className="mb-6 text-lg font-bold text-[#163B63]">
            بيانات المشارك
          </h2>

          <div className="mb-5">
            <label className={labelClass}>رقم القيد</label>
            <input
              value={employee?.empn || ""}
              readOnly
              className={`${inputClass} bg-slate-100`}
            />
          </div>

          <div className="grid gap-4 md:grid-cols-3">
            <div>
              <label className={labelClass}>الاسم</label>
              <input
                value={employee?.ename || ""}
                readOnly
                className={`${inputClass} bg-slate-100`}
              />
            </div>

            <div>
              <label className={labelClass}>المسمى الوظيفي</label>
              <input
                value={employee?.nwazifa || employee?.code_wazifa || ""}
                readOnly
                className={`${inputClass} bg-slate-100`}
              />
            </div>

            <div>
              <label className={labelClass}>الإدارة</label>
              <input
                value={employee?.edara_text || ""}
                readOnly
                className={`${inputClass} bg-slate-100`}
              />
            </div>
          </div>

          <div className="mt-5">
            <label className={labelClass}>
              الموقع (مكتب / موقع تشغيلي)
            </label>
            <input
              name="location"
              value={form.location}
              onChange={updateField}
              placeholder="أدخل الموقع"
              className={inputClass}
            />
          </div>

          <div className="mt-6">
            <div className="mb-4 flex flex-wrap items-center gap-6">
              <span className="text-sm font-bold text-[#163B63]">
                نوع الترشيح:
              </span>

              <label className="flex cursor-pointer items-center gap-2 text-sm text-slate-600">
                <input
                  type="radio"
                  name="submissionType"
                  value="1"
                  checked={submissionType === "1"}
                  onChange={() => setSubmissionType("1")}
                  className="accent-[#163B63]"
                />
                فردي
              </label>

              <label className="flex cursor-pointer items-center gap-2 text-sm text-slate-600">
                <input
                  type="radio"
                  name="submissionType"
                  value="2"
                  checked={submissionType === "2"}
                  onChange={() => setSubmissionType("2")}
                  className="accent-[#163B63]"
                />
                فريق عمل
              </label>
            </div>

            {submissionType === "2" && (
              <div>
                <label className={labelClass}>
                  أعضاء الفريق ومسمياتهم

                <textarea

        {message && (

        {/* Actions */}
        <div className="flex flex-wrap justify-end gap-3">
          <Link
            href="/employee/submissions"
            className="rounded-lg border border-slate-200 bg-white px-6 py-3 text-sm font-semibold text-slate-600 hover:bg-slate-50"
          >
            إلغاء
          </Link>

          <button
            type="submit"
            disabled={submitting || loadingOptions}
            className="inline-flex min-w-40 items-center justify-center gap-2 rounded-lg bg-[#163B63] px-7 py-3 text-sm font-bold text-white transition hover:bg-[#0D2B4B] disabled:cursor-not-allowed disabled:opacity-60"

                <LoaderCircle size={18} className="animate-spin" />
                جاري الحفظ...
              </>
            ) : (
              <>
                <Save size={18} />
                حفظ المشاركة
              </>
            )}
          </button>
        </div>
      </form>
    </div>
  );
}

function formatDate(value) {
  if (!value) return "—";
  const date = new Date(value);
  return Number.isNaN(date.getTime())
    ? "—"
    : date.toLocaleDateString("ar-EG", { year: "numeric", month: "long", day: "numeric" });
}

function formatSize(value) {
  const size = Number(value || 0);
  if (size < 1024) return `${size} بايت`;
  if (size < 1024 * 1024) return `${(size / 1024).toFixed(0)} كيلوبايت`;
  return `${(size / (1024 * 1024)).toFixed(1)} ميجابايت`;
}

export default function EmployeeSubmissionDetailsPage() {
  const { id } = useParams();
  const router = useRouter();
  const [submission, setSubmission] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [downloadingFile, setDownloadingFile] = useState(null);
  const [fileError, setFileError] = useState("");

  useEffect(() => {
    let active = true;
    const token = sessionStorage.getItem("auth_token");

    if (!token) {
      router.replace("/login");
      return () => {
        active = false;
      };
    }

    fetch(`${API_URL}/employee/submissions/${encodeURIComponent(id)}`, {
      headers: { Accept: "application/json", Authorization: `Bearer ${token}` },
      cache: "no-store",
    })
      .then(async (response) => ({ response, data: await response.json() }))
      .then(({ response, data }) => {
        if (!active) return;
        if (response.status === 401) {
          clearAuthSession("employee");
          router.replace("/login");
          return;
        }
        if (!response.ok || !data.success) {
          throw new Error(data.message || "تعذر تحميل تفاصيل المشاركة.");
        }
        setSubmission(data.submission);
      })
      .catch((loadError) => {
        if (active) setError(loadError.message || "حدث خطأ أثناء تحميل التفاصيل.");
      })
      .finally(() => {
        if (active) setLoading(false);
      });

    return () => {
      active = false;
    };
  }, [id, router]);

  async function downloadFile(file) {
    const token = sessionStorage.getItem("auth_token");
    if (!token) {
      router.replace("/login");
      return;
    }

    setDownloadingFile(file.id);
    setFileError("");
    try {
      const response = await fetch(`${API_URL}/employee/submission-files/${file.id}`, {
        headers: { Authorization: `Bearer ${token}` },
        cache: "no-store",
      });
      if (response.status === 401) {
        clearAuthSession("employee");
        router.replace("/login");
        return;
      }
      if (!response.ok) {
        const result = await response.json().catch(() => ({}));
        throw new Error(result.message || "تعذر تنزيل المرفق.");
      }

      const blobUrl = URL.createObjectURL(await response.blob());
      const link = document.createElement("a");
      link.href = blobUrl;
      link.download = file.name || "attachment";
      document.body.appendChild(link);
      link.click();
      link.remove();
      URL.revokeObjectURL(blobUrl);
    } catch (downloadError) {
      setFileError(downloadError.message || "تعذر تنزيل المرفق.");
    } finally {
      setDownloadingFile(null);
    }
  }

    );

const API_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://127.0.0.1:8000/api";

async function criteriaRequest(url, options = {}) {
  const token = sessionStorage.getItem("hr_token");

  const response = await fetch(`${API_URL}${url}`, {
    ...options,
    headers: {
      Accept: "application/json",
      ...(options.body ? { "Content-Type": "application/json" } : {}),
      Authorization: `Bearer ${token}`,
      ...options.headers,
    },
    cache: "no-store",
  });

  const result = await response.json();

  if (response.status === 401) {
    sessionStorage.removeItem("hr_token");
    sessionStorage.removeItem("hr_user");
    window.location.href = "/login";
    throw new Error("انتهت جلسة تسجيل الدخول.");
  }

  if (!response.ok || !result.success) {
    throw new Error(result.message || "حدث خطأ أثناء تنفيذ العملية.");
  }

  return result;
}

const emptyForm = {
  Id: null,
  NameAr: "",
  SortOrder: 1,
  IsActive: true,
};

export default function HRCriteriaPage() {
  const [criteria, setCriteria] = useState([]);
  const [form, setForm] = useState(emptyForm);
  const [editing, setEditing] = useState(false);
  const [showForm, setShowForm] = useState(false);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");

  const loadCriteria = useCallback(async () => {
    try {
      setLoading(true);
      setError("");

      const result = await criteriaRequest("/hr/criteria");
      setCriteria(Array.isArray(result.data) ? result.data : []);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    loadCriteria();
  }, [loadCriteria]);

  function openAdd() {
    setForm({
      ...emptyForm,
      SortOrder: criteria.length + 1,
    });
    setEditing(false);
    setShowForm(true);
    setMessage("");
    setError("");
  }

  function openEdit(item) {
    setForm({
      Id: item.Id,
      NameAr: item.NameAr,
      SortOrder: item.SortOrder,
      IsActive: Boolean(item.IsActive),
    });
    setEditing(true);
    setShowForm(true);
    setMessage("");
    setError("");
  }

  async function handleSave(e) {
    e.preventDefault();

    if (!form.NameAr.trim()) {
      setError("من فضلك أدخلي اسم المعيار.");
      return;
    }

    try {
      setSaving(true);
      setError("");
      setMessage("");

      const payload = {
        NameAr: form.NameAr.trim(),
        SortOrder: Number(form.SortOrder),
        IsActive: Boolean(form.IsActive),
      };

      if (editing) {
        await criteriaRequest(`/hr/criteria/${form.Id}`, {
          method: "PUT",
          body: JSON.stringify(payload),
        });
        setMessage("تم تحديث المعيار بنجاح.");
      } else {
        await criteriaRequest("/hr/criteria", {
          method: "POST",
          body: JSON.stringify(payload),
        });
        setMessage("تمت إضافة المعيار بنجاح.");
      }

      setShowForm(false);
      await loadCriteria();
    } catch (err) {
      setError(err.message);
    } finally {
      setSaving(false);
    }
  }

  async function toggleActive(item) {
    try {

      setMessage("");

        body: JSON.stringify({
          NameAr: item.NameAr,
          SortOrder: Number(item.SortOrder),
          IsActive: !Boolean(item.IsActive),
        }),

          </p>

          <button
            onClick={loadCriteria}
            disabled={loading}

            onClick={openAdd}
            className="flex items-center gap-2 rounded-xl bg-[#163B63] px-5 py-3 text-sm font-semibold text-white hover:bg-[#0D2B4B]"

const API_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://127.0.0.1:8000/api";

async function settingsRequest(url, options = {}) {
  const token = sessionStorage.getItem("hr_token");

  const response = await fetch(`${API_URL}${url}`, {
    ...options,
    headers: {
      Accept: "application/json",
      ...(options.body ? { "Content-Type": "application/json" } : {}),
      Authorization: `Bearer ${token}`,
      ...options.headers,
    },
    cache: "no-store",
  });

  const result = await response.json();

  if (response.status === 401) {
    sessionStorage.removeItem("hr_token");
    sessionStorage.removeItem("hr_user");
    window.location.href = "/login";
    throw new Error("انتهت جلسة تسجيل الدخول.");
  }

  if (!response.ok || !result.success) {
    throw new Error(result.message || "حدث خطأ أثناء تنفيذ العملية.");
  }

  return result;
}

function toDateInput(value) {
  if (!value) return "";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "";

  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

      try {
        setLoading(true);
        setError("");

        const result = await getHRSubmissions(filters);

        const submissions = Array.isArray(result.submissions)
          ? result.submissions
          : Array.isArray(result)

            : [];

          ? result.departments
          : [];

          submissions,
          departments,
        });

        if (!cancelled) {
          setError(
            err.message || "تعذر تحميل المشاركات."
          );
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

      fetchSubmissions();

      ...current,
      [name]: value,

    setFilters({ ...initialFilters });

  AlertCircle,
} from "lucide-react";

const API_URL =
  process.env.NEXT_PUBLIC_API_URL || "http://127.0.0.1:8000/api";

function getToken() {
  if (typeof window === "undefined") return null;
  return sessionStorage.getItem("hr_token");
}

function StorySection({ title, children }) {
  return (
    <section className="rounded-2xl border border-[#D5DFEC] bg-white p-6 md:p-8">
      <h2 className="mb-5 text-xl font-bold text-[#102F5F]">
        {title}
      </h2>

      <div className="whitespace-pre-wrap leading-8 text-slate-700">
        {children || "لم يتم إدخال بيانات."}
      </div>
    </section>
  );
}

export default function HRSubmissionDetails() {
  const params = useParams();
  const router = useRouter();
  const id = params.id;

  const [previewFile, setPreviewFile] = useState(null);
  const [submission, setSubmission] = useState(null);

  const [stars, setStars] = useState(0);
  const [notes, setNotes] = useState("");

  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [openingFile, setOpeningFile] = useState(null);

  const [error, setError] = useState("");
  const [message, setMessage] = useState("");

  // =========================================
  // Load submission details
  // =========================================
  const loadSubmission = useCallback(async () => {
    const token = getToken();

    if (!token) {
      router.replace("/login");
      return;
    }

    try {
      setLoading(true);
      setError("");

      const response = await fetch(
        `${API_URL}/hr/submissions/${id}`,
        {
          method: "GET",
          headers: {
            Accept: "application/json",
            Authorization: `Bearer ${token}`,
          },
          cache: "no-store",
        }
      );

      const result = await response.json();

      if (response.status === 401) {
        sessionStorage.removeItem("hr_token");
        sessionStorage.removeItem("hr_user");
        router.replace("/login");
        return;
      }

      if (!response.ok || !result.success) {
        throw new Error(
          result.message || "تعذر تحميل تفاصيل المشاركة."
        );
      }

      setSubmission(result.submission);
      setStars(Number(result.submission.my_review?.Stars || 0));
      setNotes(result.submission.my_review?.Notes || "");
    } catch (err) {
      console.error("Load Submission Error:", err);
      setError(err.message || "حدث خطأ أثناء تحميل المشاركة.");
    } finally {
      setLoading(false);
    }
  }, [id, router]);

  // Load only once per submission ID
  useEffect(() => {
    loadSubmission();
  }, [loadSubmission]);

  // =========================================
  // Save or update evaluation
  // =========================================
  async function handleSaveEvaluation(e) {
    e.preventDefault();

    if (stars < 1 || stars > 5) {
      setError("اختاري التقييم من نجمة إلى خمس نجوم.");
      return;
    }

    try {
      setSaving(true);
      setError("");
      setMessage("");

      const token = getToken();

      if (!token) {
        router.replace("/login");
        return;
      }

      const response = await fetch(
        `${API_URL}/hr/submissions/${id}/evaluate`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            Accept: "application/json",
            Authorization: `Bearer ${token}`,
          },
          body: JSON.stringify({
            stars,
            notes,
          }),
        }
      );

      const result = await response.json();

      if (response.status === 401) {
        sessionStorage.removeItem("hr_token");
        sessionStorage.removeItem("hr_user");
        router.replace("/login");
        return;
      }

      if (!response.ok || !result.success) {
        throw new Error(result.message || "تعذر حفظ التقييم.");
      }

      setMessage("تم حفظ التقييم بنجاح.");

      await loadSubmission();
    } catch (err) {
      console.error("Save Evaluation Error:", err);
      setError(err.message || "حدث خطأ أثناء حفظ التقييم.");
    } finally {
      setSaving(false);
    }
  }

  // =========================================
  // Preview attachment inside the website
  // =========================================
  async function handleOpenFile(file) {
    try {
      setOpeningFile(file.id);
      setError("");
      setMessage("");

      const token = getToken();

      if (!token) {
        router.replace("/login");
        return;
      }

      // Close any previous preview before opening another file
      if (previewFile?.url) {
        URL.revokeObjectURL(previewFile.url);
        setPreviewFile(null);
      }

      // Fetch the protected file using the HR token
      const response = await fetch(file.url, {
        method: "GET",
        headers: {
          Authorization: `Bearer ${token}`,
          Accept: "*/*",
        },
        cache: "no-store",
      });

      if (response.status === 401) {
        sessionStorage.removeItem("hr_token");
        sessionStorage.removeItem("hr_user");
        router.replace("/login");
        return;
      }

      if (!response.ok) {
        let result = {};

        try {
          result = await response.json();
        } catch {
          // Response may not be JSON
        }

        throw new Error(
          result.message || "تعذر تحميل المرفق."
        );
      }

      // Read the response as a Blob instead of navigating to file.url
      const blob = await response.blob();

      if (!blob.size) {
        throw new Error("الملف فارغ أو تعذر قراءته.");
      }

      // Create a temporary URL for the preview modal
      const objectUrl = URL.createObjectURL(blob);

      setPreviewFile({
        id: file.id,
        name: file.name,
        blob,
        url: objectUrl,
      });
    } catch (err) {
      console.error("File Preview Error:", err);
      setError(err.message || "تعذر فتح المرفق.");
    } finally {
      setOpeningFile(null);
    }
  }

  // =========================================
  // Close preview and release temporary URL
  // =========================================
  function handleClosePreview() {
    if (previewFile?.url) {
      URL.revokeObjectURL(previewFile.url);
    }

    setPreviewFile(null);
  }

  // Release temporary URL if the page is unmounted
  useEffect(() => {
    return () => {
      if (previewFile?.url) {
        URL.revokeObjectURL(previewFile.url);
      }
    };
  }, [previewFile]);

  // =========================================
  // Loading state
  // =========================================
  if (loading) {
    return (
      <main
        dir="rtl"
        className="flex min-h-[60vh] items-center justify-center bg-[#F2F4F8]"
      >
        <div className="flex items-center gap-3 text-[#163B63]">
          <RefreshCw size={22} className="animate-spin" />
          جاري تحميل تفاصيل المشاركة...
        </div>
      </main>
    );
  }

  // =========================================
  // Error state
  // =========================================
  if (error && !submission) {
    return (
      <main
        dir="rtl"
        className="min-h-screen bg-[#F2F4F8] px-5 py-10 md:px-12"
      >
        <div className="mx-auto max-w-3xl rounded-2xl border border-red-200 bg-white p-8 text-center">
          <AlertCircle
            className="mx-auto mb-4 text-red-500"
            size={32}
          />

          <p className="mb-5 text-red-600">{error}</p>

          <button
            type="button"
            onClick={loadSubmission}
            className="cursor-pointer rounded-xl bg-[#163B63] px-6 py-3 text-white transition hover:bg-[#0D2B4B]"
          >
            إعادة المحاولة
          </button>
        </div>
      </main>
    );
  }

  if (!submission) return null;

  // =========================================
  // Main page
  // =========================================
  return (
    <main
      dir="rtl"
      className="min-h-screen bg-[#F2F4F8] px-5 py-8 md:px-10"
    >
      <div className="mx-auto max-w-[1600px]">

        {/* Header */}
        <div className="mb-7">
          <Link
            href={`/hr/employees?department=${submission.department_id}`}
            className="mb-4 inline-flex items-center gap-2 text-sm text-[#163B63] hover:underline"
          >
            <ArrowRight size={17} />
            العودة إلى المشاركين
          </Link>

          <h1 className="text-2xl font-bold text-[#102F5F]">
            تفاصيل المشاركة والتقييم
          </h1>
        </div>

        {/* Employee and story information */}
        <div className="grid grid-cols-1 items-start gap-6 lg:grid-cols-[minmax(0,1fr)_390px]">

          {/* Main content */}
          <div className="space-y-6">

            {/* Submission information */}
            <section className="rounded-2xl border border-[#D5DFEC] bg-white p-6 md:p-8">
              <h2 className="mb-6 text-xl font-bold text-[#102F5F]">
                {submission.title}
              </h2>

              <div className="grid grid-cols-1 gap-5 text-sm sm:grid-cols-2">

                <div className="flex items-center gap-3 text-slate-700">
                  <UserRound size={19} className="text-[#163B63]" />

                  <span>
                    <strong>اسم الموظف:</strong>{" "}
                    {submission.employee_name}
                  </span>
                </div>

                <div className="flex items-center gap-3 text-slate-700">
                  <span className="font-semibold text-[#163B63]">
                    #
                  </span>

                  <span>
                    <strong>الرقم الوظيفي:</strong>{" "}
                    {submission.empn}
                  </span>
                </div>

                <div className="flex items-center gap-3 text-slate-700">
                  <Building2 size={19} className="text-[#163B63]" />

                  <span>
                    <strong>الإدارة:</strong>{" "}
                    {submission.department_name || "—"}
                  </span>
                </div>

                <div className="flex items-center gap-3 text-slate-700">
                  <CalendarDays
                    size={19}
                    className="text-[#163B63]"
                  />

                  <span>
                    <strong>تاريخ التقديم:</strong>{" "}
                    {submission.created_at
                      ? new Date(
                          submission.created_at
                        ).toLocaleDateString("ar-EG")
                      : "—"}
                  </span>
                </div>

                <div className="flex items-center gap-3 text-slate-700">
                  <FileText size={19} className="text-[#163B63]" />

                  <span>
                    <strong>نوع الترشيح:</strong>{" "}
                    {submission.submission_type_name}
                  </span>
                </div>

                <div className="flex items-center gap-3 text-slate-700">
                  <MapPin size={19} className="text-[#163B63]" />

                  <span>
                    <strong>الموقع:</strong>{" "}
                    {submission.location || "—"}
                  </span>
                </div>
              </div>

              {submission.team_members && (
                <div className="mt-5 border-t border-slate-100 pt-5 text-sm text-slate-700">
                  <strong>أعضاء الفريق:</strong>

                  <p className="mt-2 whitespace-pre-wrap">
                    {submission.team_members}
                  </p>
                </div>
              )}

              <div className="mt-6 border-t border-slate-100 pt-5">
                <h3 className="mb-3 font-bold text-[#102F5F]">
                  معايير المشاركة
                </h3>

                <div className="flex flex-wrap gap-2">
                  {submission.criteria?.length ? (
                    submission.criteria.map((criterion) => (
                      <span
                        key={criterion.Id}
                        className="rounded-full bg-[#EAF0F8] px-4 py-2 text-sm text-[#163B63]"
                      >
                        {criterion.NameAr}
                      </span>
                    ))
                  ) : (
                    <span className="text-sm text-slate-400">
                      لا توجد معايير مسجلة.
                    </span>
                  )}
                </div>
              </div>
            </section>

            {/* Story sections */}
            <StorySection title="التحدي">
              {submission.challenge}
            </StorySection>

            <StorySection title="الجهود والحل">
              {submission.efforts}
            </StorySection>

            <StorySection title="النتائج بالأرقام">
              {submission.results}
            </StorySection>

            <StorySection title="الأثر">
              {submission.impact}
            </StorySection>

            <StorySection title="ملهمة للنشر">
              {submission.inspiration}
            </StorySection>

            {/* Attachments */}
            <section className="rounded-2xl border border-[#D5DFEC] bg-white p-6 md:p-8">
              <h2 className="mb-6 text-xl font-bold text-[#102F5F]">
                المرفقات
              </h2>

              {submission.files?.length ? (
                <div className="space-y-3">
                  {submission.files.map((file) => (
                    <div
                      key={file.id}
                      className="flex flex-wrap items-center justify-between gap-4 border-b border-slate-100 py-4 last:border-0"
                    >
                      <div className="flex min-w-0 items-center gap-3">
                        <FileText
                          className="shrink-0 text-[#163B63]"
                          size={23}
                        />

                        <div className="min-w-0">
                          <p className="break-all font-medium text-slate-700">
                            {file.name}
                          </p>

                          <p className="mt-1 text-xs text-slate-500">
                            {file.type_name || "مرفق"}

                            {file.size
                              ? ` • ${(file.size / 1024 / 1024).toFixed(2)} MB`
                              : ""}
                          </p>
                        </div>
                      </div>

                      <button
                        type="button"
                        onClick={() => handleOpenFile(file)}
                        disabled={openingFile === file.id}
                        className="inline-flex cursor-pointer items-center gap-2 rounded-lg border border-[#163B63] px-5 py-2.5 text-sm text-[#163B63] transition hover:bg-[#163B63] hover:text-white disabled:cursor-not-allowed disabled:opacity-50"
                      >
                        {openingFile === file.id ? (
                          <RefreshCw
                            size={16}
                            className="animate-spin"
                          />
                        ) : (
                          <Eye size={16} />
                        )}

                        {openingFile === file.id
                          ? "جاري التحميل..."
                          : "معاينة"}
                      </button>
                    </div>
                  ))}
                </div>
              ) : (
                <p className="py-5 text-center text-slate-500">
                  لا توجد مرفقات.
                </p>
              )}
            </section>
          </div>

          {/* Evaluation Panel */}
          <aside className="lg:sticky lg:top-6">
            <form
              onSubmit={handleSaveEvaluation}
              className="rounded-2xl border border-[#D5DFEC] bg-white p-6 md:p-7"
            >
              <h2 className="mb-7 text-xl font-bold text-[#102F5F]">
                تقييم المشاركة
              </h2>

              <p className="mb-4 text-sm text-slate-600">
                اختاري التقييم من نجمة إلى خمس نجوم
              </p>

              <div
                className="mb-8 flex items-center justify-center gap-2"
                dir="ltr"
              >
                {[1, 2, 3, 4, 5].map((value) => (
                  <button
                    key={value}
                    type="button"
                    onClick={() => setStars(value)}
                    aria-label={`تقييم ${value} من 5`}
                    className="cursor-pointer rounded-md p-1 transition hover:scale-110"
                  >
                    <Star
                      size={35}
                      strokeWidth={1.8}
                      className={
                        value <= stars
                          ? "fill-amber-500 text-amber-600"
                          : "text-amber-600"
                      }
                    />
                  </button>
                ))}
              </div>

              <div className="mb-8 text-center text-sm font-semibold text-amber-700">
                {stars > 0
                  ? `${stars} من 5 نجوم`
                  : "لم يتم اختيار تقييم"}
              </div>

              <label className="mb-2 block text-sm font-medium text-slate-700">
                ملاحظات التقييم
              </label>

              <textarea
                value={notes}
                onChange={(e) => setNotes(e.target.value)}
                rows={6}
                maxLength={4000}
                placeholder="اكتبي ملاحظاتك حول المشاركة..."
                className="w-full resize-y rounded-xl border border-[#C9D4E3] p-3 text-sm outline-none transition focus:border-[#163B63] focus:ring-2 focus:ring-[#163B63]/10"
              />

              {error && (
                <div className="mt-4 flex items-start gap-2 rounded-lg bg-red-50 p-3 text-sm text-red-600">
                  <AlertCircle
                    size={17}
                    className="mt-0.5 shrink-0"
                  />

                  {error}
                </div>
              )}

              {message && (
                <div className="mt-4 flex items-center gap-2 rounded-lg bg-green-50 p-3 text-sm text-green-700">
                  <CheckCircle2 size={17} />
                  {message}
                </div>
              )}

              <button
                type="submit"
                disabled={saving || stars === 0}
                className="mt-6 flex h-12 w-full cursor-pointer items-center justify-center gap-2 rounded-xl bg-[#102F5F] font-semibold text-white transition hover:bg-[#0B2449] disabled:cursor-not-allowed disabled:opacity-50"
              >
                {saving ? (
                  <>
                    <RefreshCw
                      size={18}
                      className="animate-spin"
                    />
                    جاري حفظ التقييم...
                  </>
                ) : (
                  "حفظ التقييم"
                )}
              </button>

              {submission.my_review && (
                <p className="mt-4 text-center text-xs text-slate-500">
                  يمكنك تعديل تقييمك وحفظه مرة أخرى.
                </p>
              )}
            </form>
          </aside>
        </div>

        {/* File Preview Modal */}
        {previewFile && (
          <FilePreviewModal
            file={previewFile}
            onClose={handleClosePreview}
          />
        )}
      </div>
    </main>
  );
}
